\section*{Simplify the following expressions.}

\begin{enumerate}
\item[(a)]
\[
\frac{(2x^{-3}y^4)(6x^5y^{-2})}{3x^2y}
\]

\item[(b)]
\[
\left(\frac{-3x^2y^{-3}}{9x^{-1}y^2}\right)^{-1}
\]
\end{enumerate}

\section*{Expand and simplify.}

\[
(2x-3)^2-(x+4)(x-5)
\]

\section*{Polynomial Division}

Find the quotient and remainder:
\[
\frac{12x^3-7x^2+2x-9}{3x+2}
\]

\[
6x^3-24x+3x^2-12
\]

\item[(b)]
\[

\end{enumerate}

\section*{Solve the quadratic equation (do not use the quadratic formula).}

\[
2x^2-11x+12=0
\]

\section*{Application Problem}

A rock is dropped from a height of 196 feet. Its height is given by:
\[
h(t)=-16t^2+196
\]

How long will it take for the rock to hit the ground?

\section*{Domain of a Rational Function}

Find the domain in interval notation:
\[
f(x)=\frac{3x+1}{x^2-5x-14}
\]

\section*{Simplify the complex fraction.}

\[
\frac{1-\frac{2}{x}}{1+\frac{3}{x}}
\]

\section*{Add and simplify.}

\[
\frac{2}{x(x-4)}+\frac{5}{x-4}
\]

\section*{Solve the rational equation.}

\[
\frac{3}{y-2}=\frac{6}{y+4}
\]

\section*{Word Problem}

A fruit drink mixture uses 3 liters of concentrate for every 7 liters of water. If the total mixture is 120 liters, how many liters of water are needed?
