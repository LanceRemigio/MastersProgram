\begin{problem}[Spring 2025]
   Let \( n \geq 3   \) and let \( i \in \{  1,2,\dots, n \}  \). 
   \begin{enumerate}
       \item[(a)] Show that \( H = \{ \sigma \in {S}_{n} : \sigma(i) = i    \}   \) is a subgroup of \( {S}_{n} \). 
        \item[(b)] In \( {S}_{8} \), let \( \alpha = (1 \ 2 \ 3)(4 \ 5)(6 \ 7) \). Determine the number of elements in \( {C}_{{S}_{8}}(\alpha) \), the centralizer of \( \alpha  \) in \( {S}_{8} \).
   \end{enumerate}
\end{problem}

\begin{problem}[Fall 2024]
   Let \( G  \) and \( G' \) be groups and let \( \varphi : G \to G' \) be a surjective group homomorphism. 
   \begin{enumerate}
       \item[(a)] Prove that if \( T  \) is a subgroup of \( G' \), then \( \varphi^{-1}(T) \) is a subgroup of \( G \).
        \item[(b)] Suppose that \( T  \) is normal in \( G' \). Prove there exists a normal subgroup \( S \) of \( G  \) such that \( G / S  \) is isomorphic to \( G' /T  \).
   \end{enumerate}
\end{problem}

\begin{problem}[Fall 2024]
        If \( G  \) is a finite group of order \( n  \) and \( p  \) is the smallest prime dividing \( | G  |  \), then any subgroup of index \( p  \) is normal.
\end{problem}

\begin{problem}[Fall 2024]
    Let \( G  \) be a finite group and let the notation \( \text{ord}_S(\cdot) \) denote the order of an element in a group \( S \).
        Suppose that \( H  \) is a normal subgroup of \( G  \) and \( g \in G  \). Prove that \( \text{ord}_{G/H}(gH) \) divides \( \text{ord}_{G}(g) \).
\end{problem}

\begin{problem}[Fall 2023]
    \begin{enumerate}
        \item[(a)] Suppose \( G  \) is a group and \( N  \) is a normal subgroup of \( G  \). If \( K  \) is any other subgroup of \( G  \), prove that \( K / (N \cap K)  \) is isomorphic to a subgroup of \( G / N  \).
        \item[(b)] Suppose \( G  \) is a group with a normal subgroup \( N \) such that \( | N  |   \) and \( | G : N |  \) are relatively prime. If \( K  \) is a subgroup of \( G  \) such that \( | K  |  = | N  |  \), prove that \( N \cap K = K  \) and thus \( K =  N  \).
    \end{enumerate}
\end{problem}

\begin{problem}[Fall 2022]
    \begin{enumerate}
        \item[(a)] Suppose \( G  \) is a group of order \(  p^{n} \). Prove that \( Z(G) \) is nontrivial.
        \item[(b)] Suppose \( G  \) is a \textbf{nonabelian} group of order \( p^{3} \). Prove that \( G / Z(G) \cong {\Z}_{p} \times {\Z}_{p} \). 
    \end{enumerate}
\end{problem}

\begin{problem}[Fall 2022]
    Let \( G  \) be a finite group, and let \( p  \) be the smallest prime divisor of \( | G  |  \). Suppose \( G  \) has a subgroup \( H  \) that has index \( p  \) in \( G  \). Prove that \( H  \) is normal in \( G  \).
\end{problem}
