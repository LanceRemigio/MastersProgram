\section{Spring 2025}

\begin{problem}
   Let \( n \geq 3  \) and let \( i \in \{ 1,2,\dots, n \}  \). 
   \begin{enumerate}
       \item[(a)] Show that \( H = \{ \sigma \in {S}_{n} : \sigma (i) = i  \}   \) is a subgroup of \( {S}_{n} \).
        \item[(b)] In \( {S}_{8} \), let \( \alpha = (123)(45)(67) \). Determine the number of elements in \( {C}_{{S}_{8}}(\alpha) \), the centralizer of \( \alpha \) in \( {S}_{8} \).
   \end{enumerate}
\end{problem}

\begin{problem}
    \begin{enumerate}
        \item[(a)] Let \( G  \) be a group and let \( x \in G  \) with \( | x  |  = n  \). Prove 
            \[ x^{k} = e \iff n \mid k. \]
        \item[(b)] Prove that, in a finite Abelian group, for any \( x,y \in G  \), we have 
            \[  | x y  |  \mid \text{lcm}(| x  | , | y | ). \]
    \end{enumerate}
\end{problem}

\section{Fall 2024}

\begin{problem}
    Let \( G   \) and \( G' \) be groups and let \( \varphi : G \to G' \) be a surjective group homomorphism. Prove that if \( T  \) is a subgroup of \( G' \), then \( \varphi^{-1}(T) \) is a subgroup of \( G  \).
\end{problem}

\section{Spring 2024}

\begin{problem}
    Let \( G \) be a group and let \( a \in G  \) such that \( | a  |  \) is finite. 
    \begin{enumerate}
        \item[(a)] Let \( k \in \Z  \). Prove that \( a^{k} = e  \) if and only if \( | a  |   \) divides \( k  \).
        \item[(b)] Let \( {G}_{1} \) and \( {G}_{2} \) be finite groups and let \( (a,b) \in {G}_{1} \times {G}_{2}  \). Prove \( | (a,b) |  = \text{lcm}(| a | , | b | ) \).
    \end{enumerate}
\end{problem}

\section{Fall 2023}

\begin{problem}
   Let \( {S}_{n} \) be the symmetric group of order \( n  \).   
   \begin{enumerate}
       \item[(a)] Find the smallest \( n \) for which \( {S}_{n} \) contains an element of order \( 12 \). Justify your answer. 
        \item[(b)] Does \( {S}_{8} \) contain an element of order \( 14 \)? Justify your answer?
        \item[(c)] Find an element \( \tau  \) in \( {S}_{8} \) of maximal order. Justify your answer. 
        \item[(d)] How many conjugates does \( \sigma = (12)(34)(56)  \) have in \( {S}_{8} \)?
        \item[(e)] Let \( \sigma \) be as in part (d). Determine the order of the centralizer \( {C}_{{S}_{8}}(\sigma) \) in \( {S}_{8} \).
   \end{enumerate} 
\end{problem}

\section{Fall 2022}

\begin{problem}
    Let \( G  \) be a group. 
    \begin{enumerate}
        \item[(a)] For \( x,y \in G  \), we define \( x \sim y  \) if and only if \( x = y  \) or \( x = y^{-1} \). Prove \( \sim  \) is an equivalence relation on \( G  \). 
        \item[(b)] If \( G  \) has even order, prove that the number of elements of \( G  \) of order \( 2  \) is odd.
    \end{enumerate}
\end{problem}

\section{Fall 2021}

\begin{problem}
    Consider the permutation \( \alpha = (12345)(678) \) in \( {S}_{8} \). 
    \begin{enumerate}
        \item[(a)] Prove that \( {C}_{{S}_{8}}(\alpha) = \langle \alpha \rangle \).
        \item[(b)] Find a premutation \( \alpha \in {S}_{8} \) such that \( \sigma \alpha \sigma^{-1} = \alpha^{-1} \).
    \end{enumerate}
\end{problem}
